```powershell
python hardware/scripts/check_clock_styles_device.py --port COM8 --bin build/clock-style-update-20261030/firmware/yiyi_desk.ino.bin --out build/clock-style-update-20261030/device --host build/clock-style-update-20261030/host-clocks
```
